rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
 match /profiles/{uid} {
 function owner() { return request.auth != null && request.auth.uid == uid && request.auth.token.firebase.sign_in_provider != 'anonymous'; }
 allow get, delete: if owner();
 allow create, update: if owner() && request.resource.data.keys().hasAll(['displayName','level']) && request.resource.data.keys().hasOnly(['displayName','level']) && request.resource.data.displayName is string && request.resource.data.displayName.size() > 0 && request.resource.data.displayName.size() <= 50 && request.resource.data.level in ['beginner','player','advanced','master'];
 }
 match /trainingRatings/{uid}/modes/{mode} {
 function owner() { return request.auth != null && request.auth.uid == uid && request.auth.token.firebase.sign_in_provider != 'anonymous' && mode in ['rapid','blitz','untimed']; }
 allow get, delete: if owner();
 allow create, update: if owner() && request.resource.data.keys().hasAll(['elo','games']) && request.resource.data.keys().hasOnly(['elo','games']) && request.resource.data.elo is int && request.resource.data.elo >= 100 && request.resource.data.games is int && request.resource.data.games >= 0;
 }
// Insert inside match /databases/{database}/documents, before its closing brace.
match /swissLobby/{code} {
  function signed() { return request.auth != null && request.auth.token.email_verified == true && request.auth.token.firebase.sign_in_provider != 'anonymous'; }
  function validMember(m) { return m is map && m.keys().hasOnly(['name']) && m.name is string && m.name.size() > 0 && m.name.size() <= 50; }
  allow get: if signed();
  allow create: if signed() && code.matches('SH[0-9]{6}')
    && request.resource.data.keys().hasOnly(['host','name','capacity','wait','time','rounds','created','members','rest'])
    && request.resource.data.host == request.auth.uid
    && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60
    && request.resource.data.capacity is int && request.resource.data.capacity >= 2 && request.resource.data.capacity <= 44
    && request.resource.data.wait is int && request.resource.data.wait >= 3 && request.resource.data.wait <= 20
    && request.resource.data.time in ['1+0','3+0','3+2','5+0','5+3','10+0','10+5','15+10','30+0']
    && request.resource.data.rounds is int && request.resource.data.rounds >= 1 && request.resource.data.rounds <= 10 && request.resource.data.rounds < request.resource.data.capacity
    && request.resource.data.rest in [0,30,60,120,180,240,300]
    && request.resource.data.created == request.time
    && request.resource.data.members is map && request.resource.data.members.keys().hasOnly([request.auth.uid]) && validMember(request.resource.data.members[request.auth.uid]);
  allow update: if signed() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members'])
    && request.resource.data.members is map && request.resource.data.members.size() <= resource.data.capacity
    && request.resource.data.members.diff(resource.data.members).affectedKeys().hasOnly([request.auth.uid])
    && (!(request.auth.uid in request.resource.data.members) || (request.time < resource.data.created + duration.value(resource.data.wait * 60, 's') && validMember(request.resource.data.members[request.auth.uid])));
}

 }
}
